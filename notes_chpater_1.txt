// js ke ander ise 2 major types ke ander divide kiya gyaa hai.
// "Primitives" data types and "Reference" data types.

// Primitives Data Types: - aisi saari values jinko copy karne par tumhe ek real copy mil jaaye.
// e.g. string, number, boolean, null, undefined, symbol, bigint


let a = 12; // iske pas apna 12 hai
let b = a; // iske pas apna 12 hai
a = a + 4; // ye check karne k liye is wale a ke wajah se result 16 aayega
/*
output:
a
16
b
12
*/


// References Data Types: - inko copy karne par real copy nahi milegi but aapko reference milega parent ka.
// e.g. arrays, objects, functions
//      []        {}        ()

let a = [1,2,3];
let b = a;

b.pop(); // array se last num remove karne liye use hota hai

/*
output:
a
(2) [1, 2]
b
(2) [1, 2]
*/
// so output se humen pata chalta hai kiy b se value ko hatane se a ke value bhi remove ho ja rahi hai, jab ki Primitives Data Types me aisa nahi tha.
